"use client";

import { Box, CircularProgress, Link, Typography } from "@mui/material";
import { AuthBackdrop } from "@/components/auth/auth-backdrop";
import {
  AuthCard,
  AuthCardBody,
  AuthCardFooter,
  AuthCardHeader,
} from "@/components/auth/auth-card";
import { PrimaryButton } from "@/components/ui/primary-button";
import { themeTokens } from "@/theme/tokens";
import { TextInput } from "@/components/ui/text-input";
import { useEffect, useRef, useState } from "react";
import { AlertBanner } from "@/components/ui/alert-banner";
import { useRouter } from "next/navigation";
import { useAuth } from "@/hooks/useAuth";
import { Suspense } from "react";
import { generateOtp, insertLoginSession, validateOtp } from "@/services/auth";
import { getApiErrorMessage } from "@/services/api/api-error";
import {
  getSsoBridgePath,
  getStoredSsoRequestId,
} from "@/lib/sso/sso-request";
import { AuthStatus } from "@/constants/auth-status";
import { resolveDisplayName } from "@/lib/auth/display-name";
import { markAuthVerified } from "@/context/auth-context";
 
 
const { colors, typography } = themeTokens;

function OtpContent() {


  const [otp, setOtp] = useState(["", "", "", "", "", ""]);
  const [showOtpError, setShowOtpError] = useState(false);
  const inputRefs = useRef<(HTMLInputElement | null)[]>([]);
  const router = useRouter();
  const otpError = showOtpError;
  const [showRequiredError, setShowRequiredError] = useState(false);
  const [isResending, setIsResending] = useState(false);
  const [loading, setLoading] = useState(false);
  const [systemErrorMessage, setSystemErrorMessage] = useState("");
  const {
    email,
    phone,
    otpDestination,
    otpMethod,
    loginUserName,
    email: authEmail,
    phone: authPhone,
    setAuthAccessToken,
    setAuthSessionId,
    setAuthUserId,
    setAuthUsername,
    setEmail,
    setPhone,
    setFullName,
    clearAuth,
  } = useAuth();
  const isPhone = otpMethod === "phone";
  const currentMaskedHint = isPhone ? authPhone : authEmail;
  const [maskedHint, setMaskedHint] = useState(currentMaskedHint);

  useEffect(() => {
  if (!maskedHint) {
      setMaskedHint(currentMaskedHint);
    }
  }, [currentMaskedHint, maskedHint]);

  async function handleLogin() {
    const otpValue = otp.join("");

    if (!otpValue) {
      setShowRequiredError(true);
      setShowOtpError(false);
      setSystemErrorMessage("");
      return;
    }

    if(otpValue.length < 6) {
      setShowOtpError(true);
      setShowRequiredError(false);
      setSystemErrorMessage("");
      return;
    }

    if (loading) {
      return;
    }

    setShowRequiredError(false);
    setSystemErrorMessage("");
    setLoading(true);

    try {
      const response = await validateOtp({
        UserName: loginUserName,
        Email: loginUserName,
        Destination: otpDestination,
        Provider: otpMethod === "phone" ? 2 : 1,
        Code: otpValue,
        ReturnUrl: "",
        RememberBrowser: false,
        RememberMe: true,
      });

      if (response.Status === AuthStatus.Success) {
        if (response.Token) {
          setAuthAccessToken(response.Token);
          localStorage.setItem("accessToken", response.Token);
          // Replace the short-lived MFA token in the cookie with the new full-session token.
          // The proxy reads the cookie — without this write, the old partial token stays in
          // the cookie and may be expired or refused by TokenRefreshManager on first load.
          const expires = new Date();
          expires.setDate(expires.getDate() + 7);
          document.cookie = `accessToken=${response.Token}; Path=/; Expires=${expires.toUTCString()}; SameSite=Lax`;
        }

        setAuthUsername(response.UserName);
        setEmail(response.Email ?? email);
        setPhone(response.PhoneNumber ?? phone);
        setFullName(resolveDisplayName(response));

        const tenantId = Number(process.env.NEXT_PUBLIC_TENANT_ID) || 7143;
        const userId = response.Id ? Number(response.Id) : 0;

        try {
          const loginSession = await insertLoginSession({
            UserName: loginUserName,
            Password: "",
            RememberMe: false,
            TenantId: tenantId,
            LoginSessionId: response.SessionId,
            LogOutMode: "A",
            UserId: userId,
            ExpirationTime: new Date().toISOString(),
          });

          if (loginSession?.SessionId) {
            setAuthSessionId(loginSession.SessionId);
          }
        } catch (error) {
          console.error("InsertLoginSession failed", error);
        }

        setAuthUserId?.(userId);

        // Status = 0: OTP fully verified — user is completely authenticated.
        // This is the ONLY place markAuthVerified is called for the OTP flow.
        markAuthVerified();

        const pendingSsoRequestId = getStoredSsoRequestId();
        if (pendingSsoRequestId) {
          router.push(getSsoBridgePath(pendingSsoRequestId));
          return;
        }

        router.push("/home");
      } 
      else if (response.Status === AuthStatus.PasswordExpired) {
        if (response.Token) {
          setAuthAccessToken(response.Token);
          localStorage.setItem("accessToken", response.Token);
        }
      
        router.push("/force-set-password");
      }
      else if (response.Status === AuthStatus.Failed) {
        setShowOtpError(true);
        setOtp(["", "", "", "", "", ""]);
        inputRefs.current[0]?.focus();
      }
    } catch (error) {
      setShowOtpError(false);
      setSystemErrorMessage(getApiErrorMessage(error));
      setOtp(["", "", "", "", "", ""]);
      inputRefs.current[0]?.focus();
    } finally {
      setLoading(false);
    }
  }

  async function handleResendCode(
    event: React.MouseEvent<HTMLElement>
  ) {
    event.preventDefault();
    event.stopPropagation();

    setOtp(["", "", "", "", "", ""]);
    setShowOtpError(false);
    setShowRequiredError(false);
    setSystemErrorMessage("");

    const destination = otpDestination || phone || email;

    if (!destination) {
      setSystemErrorMessage(getApiErrorMessage(null));
      return;
    }

    try {
      setIsResending(true);
      await generateOtp({
        UserName: loginUserName,
        Email: loginUserName,
        Destination: destination,
        Provider: otpMethod === "phone" ? 2 : 1,
        Code: "",
        ReturnUrl: "",
        RememberBrowser: true,
        RememberMe: true,
      });
      inputRefs.current[0]?.focus();
    } catch (error) {
      console.error(error);
      setSystemErrorMessage(getApiErrorMessage(error));
    } finally {
      setIsResending(false);
    }
  }

  return (
    <AuthBackdrop>
      <AuthCard>
        <AuthCardHeader />

        <AuthCardBody
          sx={{
            "& > :not(style) ~ :not(style)": {
              marginTop: 1,
            },
          }}
        >
          {showRequiredError && (
            <AlertBanner
              message="Please enter the verification code."
            />
          )}
          {showOtpError && (
            <AlertBanner
              message="Invalid verification code. Please try again!"
            />
          )}
          {systemErrorMessage ? (
            <AlertBanner message={systemErrorMessage} />
          ) : null}
          <Typography
            variant="h2"
            sx={{
              color: colors.text.primary,
              fontFamily: typography.fontFamily,
              fontSize: "24px",
              fontWeight: 400,
              lineHeight: "normal",
            }}
          >
            Confirm
          </Typography>

          <Typography
            sx={{
              fontSize: "11px",
              color: colors.text.primary,
              fontFamily: typography.fontFamily,
              width: {
                xs: "100%",
                sm: "100%",
                md: "340px",
              },
            }}
          >
            Please type 6-digit verification code sent to {maskedHint}.
          </Typography>

          <Box
            sx={{
              display: "flex",
              justifyContent: "space-between",
              gap: 1,
            }}
          >
            {[...Array(6)].map((_, index) => (
              <Box
                key={index}
                sx={{
                  width: 45,
                  mt: 1,

                  alignItems: "center",
                }}
              >
                <TextInput
                  name={`otp-${index}`}
                  type="text"
                  placeholder=""
                  hideHelperText
                  onKeyDown={(event) => {
                    if (event.key === "Enter") {
                      handleLogin();
                    }

                    if (event.key === "Backspace") {
                      setShowOtpError(false);
                      setShowRequiredError(false);

                      // Current box has a value -> clear it
                      if (otp[index]) {
                        const updatedOtp = [...otp];
                        updatedOtp[index] = "";
                        setOtp(updatedOtp);
                        return;
                      }

                      // Current box is empty -> move to previous box and clear it
                      if (index > 0) {
                        const updatedOtp = [...otp];
                        updatedOtp[index - 1] = "";
                        setOtp(updatedOtp);

                        inputRefs.current[index - 1]?.focus();
                      }
                    }
                  }}
                  inputRef={(el) => {
                    inputRefs.current[index] = el;
                  }}
                  value={otp[index]}
                  onChange={(value) => {
                    setShowRequiredError(false);
                    setShowOtpError(false);
 
                    if (!/^\d*$/.test(value)) {
                      return;
                    }

                     if (value.length > 1) {
                      const updatedOtp = [...otp];

                      value
                        .slice(0, 6)
                        .split("")
                        .forEach((digit, offset) => {
                          if (index + offset < 6) {
                            updatedOtp[index + offset] = digit;
                          }
                        });

                      setOtp(updatedOtp);

                      inputRefs.current[
                        Math.min(index + value.length - 1, 5)
                      ]?.focus();

                      return;
                    }
 
                    const updatedOtp = [...otp];
                    updatedOtp[index] = value.slice(0, 1);
                    setOtp(updatedOtp);
 
                    if (value && index < 5) {
                      inputRefs.current[index + 1]?.focus();
                    }
                  }}
                  sx={{
                    textAlign: "center",
 
                    "& .MuiOutlinedInput-root": {
                      backgroundColor: colors.background.otpBackground,
                      border: `1px solid ${otpError || showRequiredError ? colors.text.dangerDark : colors.border.dark}`,
                      borderRadius: "10px",
                    },
 
                    "& input": {
                      textAlign: "center",
                      fontSize: "24px",
                      fontWeight: 600,
                    },
                    "& .MuiInputBase-input": {
                      textAlign: "center",
                      fontSize: "24px",
                      fontWeight: 600,
                      padding: 0,
                    },
                    "& input[type=number]": {
                      MozAppearance: "textfield",
                    },
                    "& input[type=number]::-webkit-outer-spin-button, & input[type=number]::-webkit-inner-spin-button": {
                      WebkitAppearance: "none",
                      margin: 0,
                    },
                    "& .MuiOutlinedInput-notchedOutline": {
                      border: "none",
                    },
 
                    "& .MuiOutlinedInput-root.Mui-focused": {
                      backgroundColor: "transparent !important",
                      boxShadow: "none !important",
                    },
                    "& .MuiOutlinedInput-root:hover .MuiOutlinedInput-notchedOutline": {
                      border: "none !important",
                    },
                    "& .MuiOutlinedInput-root.Mui-focused .MuiOutlinedInput-notchedOutline": {
                      border: "none !important",
                      boxShadow: "none !important",
                    },
 
                  }}
                />
              </Box>
            ))}
          </Box>
          {showRequiredError && (
            <Typography
              sx={{
                color: colors.text.error,
                fontSize: "10px",
                fontFamily: typography.fontFamily,
                mt: 0.5,
              }}
            >
              *Required field
            </Typography>
          )}
 
          <Typography
            sx={{
              fontSize: "12px",
              color: colors.text.primary,
              fontFamily: typography.fontFamily,
              mb: 2,
            }}
          >
            Did not receive the code?{" "}
            <Link
              component="button"
              type="button"
              underline="always"
              onClick={handleResendCode}
              sx={{
                color: colors.focus.inputHover,
                fontWeight: 500,
                fontFamily: typography.fontFamily,
                textDecoration: "underline",
                textUnderlineOffset: "2px",
                pointerEvents: isResending ? "none" : "auto",
                opacity: isResending ? 0.6 : 1,
                verticalAlign: "baseline",
                cursor: "pointer",
                border: "none",
                background: "none",
                padding: 0,
                font: "inherit",
              }}
            >
              {isResending ? "Resending..." : "Resend code"}
            </Link>
          </Typography>
 
          <PrimaryButton
            onClick={handleLogin}
            disabled={loading}
            sx={{
              width: "100%",
              height: "51px",
              fontFamily: typography.fontFamily,
              backgroundColor: colors.brand.secondary,
              "&.Mui-disabled": {
                backgroundColor: colors.brand.secondary,
                color: colors.background.app,
                opacity: 1,
              },
            }}
          >
            {loading ? (
              <CircularProgress
                size={18}
                thickness={4}
                sx={{ color: "#fff", mr: 1 }}
              />
            ) : null}
            <Typography
              sx={{
                fontWeight: 400,
                fontSize: "14px",
                fontFamily: typography.fontFamily,
                lineHeight: 1,
                letterSpacing: "0%",
              }}
            >
              Login
            </Typography>
          </PrimaryButton>
 
          <PrimaryButton
            onClick={() => {
              clearAuth();
              router.push("/login");
            }}
            sx={{
              textAlign: "center",
              borderRadius: "10px",
              color: colors.action.secondaryText,
              "&:hover": {
                backgroundColor: colors.action.secondaryHover,
              },
              fontWeight: 400,
              width: "100%",
              maxWidth: "345px",
              height: "51px",
              backgroundColor: colors.action.secondary,
              cursor: "pointer",
              fontFamily: typography.fontFamily,
 
            }}
          >
            Cancel
          </PrimaryButton>
        </AuthCardBody>
        <AuthCardFooter hideBorderTop />
      </AuthCard>
    </AuthBackdrop>
  );
}
 
export default function OtpTestPage() {
  return (
    <Suspense fallback={<></>}>
      <OtpContent />
    </Suspense>
  );
}
