/** Contention opened as an FCR sub-tab (same shape as ?fcrOpen=…&fcrLabel=…). */
interface FcrOpenContention {
    examSchedulingRequestId: string;
    examRelatedContentionId: string;
    label: string;
}

interface OpenCaseTab {
    encodedId: string;
    caseId: string;
    displayCaseId?: string;
    apptDate: string;
    mode: "preview" | "edit";
    section?: string;
    appointment: TodaysDtAppointment | null;
    initialDbq?: string;
    source: "case-management" | "provider" | "facility-diagnostic";
    dbq?: QaDbqRow | null;
     originTab?: Tab;
    /** When set with section "fcr", land on this contention tab (like QA dbq). */
    fcrContention?: FcrOpenContention | null;
}

const handleFcrProceed = async (row: {
    caseId: string;
    examSchedulingRequestId?: string;
    examRelatedContentionId?: string;
    contentionName?: string;
    fcrDate?: string;
}) => {
    const displayCaseId = row.caseId.trim();
    let esrId = (row.examSchedulingRequestId ?? "").trim();
    if (!displayCaseId && !esrId) return;

    // Case APIs expect ExamSchedulingRequestId (GUID), not InternalCaseNumber.
    if (!esrId && displayCaseId && displayCaseId !== "—") {
        try {
            const results = await searchCasesByVetFedId(displayCaseId, "VetFedCaseId");
            const item = results[0];
            esrId = String(item?.Id || item?.InternalCaseNumber || "").trim();
        } catch {
            showToast("Failed to resolve case", TOAST_TYPES.ERROR);
            return;
        }
    }

    const routeId = esrId || displayCaseId;
    if (!routeId || routeId === "—") {
        showToast("Case not found", TOAST_TYPES.ERROR);
        return;
    }

    const contentionId = String(row.examRelatedContentionId ?? "").trim();
    const contentionLabel =
        String(row.contentionName ?? "").trim() || contentionId || "Contention";
    const fcrContention: FcrOpenContention | null =
        esrId && contentionId
            ? {
                  examSchedulingRequestId: esrId,
                  examRelatedContentionId: contentionId,
                  label: contentionLabel,
              }
            : null;

    openCase(
        encodeId(routeId),
        "edit",
        "final-contention-review",
        null,
        "case-management",
        displayCaseId && displayCaseId !== "—" ? displayCaseId : routeId,
        row.fcrDate || undefined,
        undefined,
        null,
        fcrContention
    );
};

                        initialSection={activeOpenCase.section}
                        initialDbq={activeOpenCase.dbq}
                        displayCaseId={activeOpenCase.displayCaseId}
                        initialFcrContention={activeOpenCase.fcrContention}
                    />
                ) : (
                    <ProviderCaseEmbed
                        key={activeCaseId}
                        encodedId={activeCaseId!}
                        readOnly={activeOpenCase.mode === "preview"}
                        initialSection={activeOpenCase.section}
                        openDiagnostic={!!activeOpenCase.appointment}
                        initialDbq={activeOpenCase.initialDbq}
                    />
                )
            )}

            {/* Queue content — always mounted so active tab, scroll position and data are preserved on return */}
            <Box sx={{ display: activeOpenCase ? "none" : "block" }}>
            {isPageLoading ? (
                <WorkQueuePageSkeleton showPageChip={openCases.length === 0} />
            ) : (
            <>
            {/* Page chip — skipped when case tab strip already shows Work Queues */}
            {openCases.length === 0 && (
                <Box sx={{ ...chipSx, mb: "20px", ...chipUnderlineSx }}>
                <Typography sx={chipTextSx}>Work Queues</Typography>
            </Box>
            )}

                    {/* Tab row */}
                    <Box
                        sx={{
                            display: "flex",
                            flexWrap: "wrap",
                            gap: "8px",
                            mb: "24px",
                        }}
                    >
                        {visibleTabs.map((tab) => {
                            const isActive = tab === effectiveTab;
                            return (
                                <Box
                                    key={tab}
                                    component="button"
                                    type="button"
                                    onClick={() => setActiveTab(tab)}
                                    sx={{
                                        display: "inline-flex",
                                        alignItems: "center",
                                        justifyContent: "center",
                                        px: "10px",
                                        py: "7.5px",
                                        borderRadius: "5px",
                                        border: `1px solid ${isActive ? wq?.tabActiveBg : wq?.tabInactiveBorder}`,
                                        bgcolor: isActive ? wq?.tabActiveBg : wq?.tabInactiveBg,
                                        color: isActive ? wq?.tabActiveText : wq?.tabInactiveText,
                                        fontSize: "10.5px",
                                        fontWeight: 400,
                                        lineHeight: "14px",
                                        whiteSpace: "nowrap",
                                        cursor: "pointer",
                                        font: "inherit",
                                        outline: "none",
                                        transition: "background-color 0.15s, color 0.15s, border-color 0.15s",
                                        "&:hover": {
                                            opacity: isActive ? 1 : 0.8,
                                        },
                                    }}
                                >
                                    {tab}
                                </Box>
                            );
                        })}
                    </Box>

                    {/* Tab content */}
                    {visibleTabs.length === 0 ? (
                        <Box
                            sx={{
                                display: "flex",
                                alignItems: "center",
                                justifyContent: "center",
                                minHeight: 320,
                            }}
                        >
                            <Typography
                                sx={{
                                    color: wq?.emptyText,
                                    fontSize: 16,
                                    fontWeight: 600,
                                    fontStyle: "italic",
                                }}
                            >
                                No work queue sections are available for your role.
                            </Typography>
                        </Box>
                    ) : effectiveTab === "Case Management" ? (
                        <CaseManagementRegular
                            onOpenCase={openCaseFromManagement}
                            onEditOrViewCase={(encodedId, mode) => openCase(encodedId, mode, undefined, null, "provider")}
                        />
                    ) : effectiveTab === "Provider" ? (
                        <ProviderHome
                            onViewCase={(enc) => openCase(enc, "preview", undefined, null, "provider")}
                            onEditCase={(enc) => openCase(enc, "edit", undefined, null, "provider")}
                        />
                    ) : effectiveTab === "Project Management Office" ? (
                        <ProjectMgmtOfficeWorkQueue />
                    ) : effectiveTab === "Facility Diagnostic" ? (
                        <FacilityDiagnostics
                            onViewCase={handleFacilityDiagnosticView}
                        />
                        ) : effectiveTab === "Contact Center" ? (
                            <ContactCenter />
                    ) : effectiveTab === "Diagnostics Admin" ? (
                        <DiagnosticsAdminWorkQueue
                        onCaseIdClick={(caseId, appointmentDate) =>
                            handleFacilityDiagnosticView({
                                caseId,
                                appointmentDate,
                            } as TodaysDtAppointment)
                        }
                        onProceed={handleDiagnosticProceed}
                    />
                                               
                    ) : effectiveTab === "QA" ? (
                        <QAWorkQueue
                            onCaseIdClick={handleQaCaseIdClick}
                            onProceed={handleQaProceed}
                            onView={handleQaView}
                            onManagerCaseIdClick={openCaseFromManagement}
                    />
            
                    ) : effectiveTab === "Final Contention Review" ? (
                        <FCRWorkQueue
                            isFcrTab={isFcrTab}
                            onCaseIdClick={(row) => openFcrCaseFromRow(row, "preview")}
                            onProceed={handleFcrProceed}
                    />

            ) : (
                <Box
                    sx={{
                        display: "flex",
                        alignItems: "center",
                        justifyContent: "center",
                        mt: "200px",
                    }}
                >
                    <Typography
                        sx={{
                            color: wq?.emptyText,
                            fontSize: 16,
                            fontWeight: 600,
                            fontStyle: "italic",
                        }}
                    >
                        Select a queue above to view cases.
                    </Typography>
                </Box>
            )}

        
            </>
            )}
            </Box>
            <DiagnosticsAdminActionDrawer
                open={diagnosticsAdminDrawer !== null}
                type={diagnosticsAdminDrawer ?? "reschedule"}
                onClose={() => {
                    setDiagnosticsAdminDrawer(null);
                    setRescheduleRow(null);
                }}
                rescheduleRow={rescheduleRow}
                isDark={isDark}
                onRescheduleSave={
                    handleDiagnosticsAdminRescheduleSave
                }
                onUploadErrorSave={
                    handleDiagnosticsAdminUploadErrorSave
                }
            />
        </Box>
    );
};

export default WorkQueues;
